\begin{frame}{The problem (from the paper)}
  \textbf{Goal:} make a humanoid robot walk on \textbf{agricultural soil} — furrowed,
  deformable ground where the feet sink and slip.
  \begin{itemize}
    \item Real soil is \textbf{not uniform}: it changes from firm \& grippy to soft \& slippery.
    \item Two things vary independently:
      \begin{itemize}
        \item \textbf{Traction} — how grippy the surface is (friction $\mu$).
        \item \textbf{Support} — how firm it is / how much the foot sinks (stiffness).
      \end{itemize}
    \item The robot must \textbf{judge how hard the current soil is} to adapt and stay upright.
  \end{itemize}
  \vfill
  \begin{block}{The paper's pitch}
    Combine traction + support into one human-readable ``\textbf{soil difficulty score}''
    using \textbf{fuzzy logic}, and use it to drive training / reduce foot slip ($>$95\% claimed).
  \end{block}
\end{frame}

\begin{frame}{What the ``fuzzy soil index'' actually is}
  A \textbf{fuzzy (Mamdani) controller}: two inputs $\to$ one difficulty score.
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Inputs: \textbf{traction} $\mu$ and \textbf{support} (stiffness).
        \item Each is graded \textit{Low / Med / High} with \textbf{smooth} (fuzzy) boundaries.
        \item Rules $\to$ soil is \textit{Easy / Moderate / Hard}.
        \item Output: one number $d\in[0.2,0.8]$ (higher = harder).
      \end{itemize}
      \textbf{Selling points:} interpretable bins, smooth interpolation, blends two factors.
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \fbox{\parbox{0.95\linewidth}{\centering
        \textbf{Two soil ``dials''}\\[4pt]
        traction $\mu$ \;\&\; support (firmness)\\[6pt]
        $\downarrow$ fuzzy rules $\downarrow$\\[6pt]
        \textbf{difficulty} = Easy / Mod / Hard}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{First impression: the fuzzy index looks convincing}
  Sweep the soil from grippy to slippery, and plot the fuzzy score $d$ against the
  \textbf{real measured difficulty}:
  \begin{center}\fig{fuzzy_validation.png}{0.97\linewidth}\end{center}
  \good{$d$ tracks difficulty almost perfectly} (slip, falls, episode length, velocity error;
  Spearman rank correlation $|\rho|\approx 0.97$ --- same ordering of soils).
  \textbf{This is the chart that makes the fuzzy index look good} --- but it only shows $d$ tracks
  \textit{one} axis; whether that beats the raw numbers is the next question.
\end{frame}

\begin{frame}{Our question}
  \begin{block}{}
    \centering \Large Does the fuzzy index \textbf{actually add value} over just using the
    \textbf{raw soil numbers}?
  \end{block}
  \vfill
  \begin{itemize}
    \item A fuzzy score is only useful if it \textbf{beats the raw measurements} it is built from.
    \item We replicate the recipe on the \textbf{Hunter} humanoid in \textbf{IsaacSim} (HumanoidVerse)
          and test the index four ways.
    \item We judge it as a \textbf{difficulty descriptor} and as a \textbf{control signal}.
  \end{itemize}
  \vfill
  \footnotesize Honest-negative friendly: we want the right answer, not a flattering one.
\end{frame}

\begin{frame}{Sets A \& B (rigid simulator)}
  \textbf{Set A — fuzzy as a label:} does $d$ track difficulty / beat raw $\mu$?
  \begin{itemize}
    \item $d$ orders soil correctly (Spearman $\rho\approx0.97$) \dots
    \item \dots but adds \bad{nothing} over raw friction $\mu$.
    \item \textbf{Why:} rigid PhysX has \textbf{no real ``firmness'' axis} — the foot can't sink,
          so the 2-dial model collapses to 1 dial (friction).
  \end{itemize}
  \vspace{4pt}
  \textbf{Set B — fuzzy drives the training curriculum:}
  \begin{itemize}
    \item fuzzy pacing $\approx$ crisp threshold $\approx$ fixed schedule (no significant difference).
  \end{itemize}
  \vfill
  \footnotesize But the $\rho\approx0.97$ chart looked so good --- what's going on? $\to$
\end{frame}

\begin{frame}{Set A/B --- the full evidence (rigid simulator)}
  \begin{center}\fig{experiments_summary.png}{0.80\linewidth}\end{center}
  \scriptsize
  \textbf{(1)} survival \good{replicates} --- paper recipe walks $8.8\times$ longer\quad
  \textbf{(2)} slip \bad{does NOT} replicate (worse, not $<11$)\\
  \textbf{(3)} reward gate \& contact compliance are \bad{inert} (overlap within noise)\quad
  \textbf{(4)} fuzzy $d$ tracks friction ($\rho{=}0.97$) but \bad{adds nothing over raw $\mu$}
\end{frame}

\begin{frame}{Set B: how the fuzzy \textit{curriculum} works}
  \textbf{Curriculum =} train from \good{easy} soil (grippy, $\mu{=}0.80$) $\to$ \bad{hard}
  (slippery, $\mu{=}0.35$). \textbf{The real question:} \textit{who decides when to move to harder soil?}
  \vspace{3pt}
  \begin{center}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{Pacing rule} & \textbf{Like\ldots} & \textbf{Move to harder soil when\ldots} \\
    \midrule
    \textbf{fixed} & a \textbf{timer}            & a set number of blocks passed \;(\textit{ignores skill}) \\
    \textbf{crisp} & a \textbf{pass/fail exam}   & skill $\ge$ threshold \;(\textit{hard yes/no at 50\%}) \\
    \textbf{fuzzy} & a \textbf{progress bar}     & graded scores (Hold/Maybe/Advance) \textbf{add up} to 1 \\
    \bottomrule
  \end{tabular}
  \end{center}
  \good{Same} ladder, \good{same} skill signal, \good{same} ``at most 1 step harder per block'' --- the
  \textbf{only} thing that changes is the decision rule.
  \begin{block}{The honest design choice}
    ``Skill'' $=$ \textbf{how long the robot survives} (not the soil score). Fuzzy's real twist is
    \textbf{accumulating a smooth score} --- otherwise it would just be ``crisp with a new name''.
  \end{block}
\end{frame}

\begin{frame}{Set B: the result --- fuzzy \textit{ties} the simple rules}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \centering \fig{setB_auc_only.png}{0.9\linewidth}
    \end{column}
    \begin{column}{0.48\textwidth}
      \footnotesize
      After training, re-test every policy across \textit{all} soils --- a fair \textit{held-out} sweep.\\[6pt]
      \textbf{Robustness AUC} $=$ one summary score $=$ how well it survives across the board
      (higher $=$ tougher everywhere):\\[5pt]
      \centering fixed \textbf{0.64} \;\textbar\; crisp \textbf{0.68} \;\textbar\; fuzzy \textbf{0.72}\\[5pt]
      \raggedright The gap is \bad{not significant} (Welch $t{=}0.45$, 3 seeds): fuzzy's higher
      average comes from \textbf{just one lucky seed}.
    \end{column}
  \end{columns}
  \begin{block}{Bottom line}
    The curriculum \good{works}, but \textbf{fuzzy $\approx$ crisp $\approx$ fixed}: in the rigid sim the
    \textbf{firmness axis is dead}, so fuzzy's blending has \bad{nothing extra to exploit}.
  \end{block}
\end{frame}

\begin{frame}{Set B: what the robustness AUC \textit{measures}}
  \begin{columns}
    \begin{column}{0.6\textwidth}
      \centering \fig{setB_robustness_curve.png}{\linewidth}
    \end{column}
    \begin{column}{0.38\textwidth}
      \footnotesize
      After training, re-test across \textit{all} soils $\to$ a \textbf{survival curve}
      (episode length $\div$ cap, vs friction $\mu$).\\[6pt]
      \textbf{Robustness AUC} $=$ \textbf{shaded area} $\div$ $\mu$-range $=$ \textbf{average survival}
      across soils:
      \[\text{AUC}=\frac{1}{\mu_{\max}-\mu_{\min}}\int \frac{\overline{\text{ep\_len}}(\mu)}{\text{cap}}\,d\mu\]
      \scriptsize (cap $=1000$ steps). One number per arm $=$ the bar on the previous slide.\\[4pt]
      \footnotesize Curves \textbf{overlap} $\Rightarrow$ AUCs tie: \textbf{.64\,/\,.68\,/\,.72}.
    \end{column}
  \end{columns}
  \begin{block}{}
    \centering A higher curve \textit{everywhere} $=$ tougher policy. Fuzzy's curve is
    \bad{not consistently above} fixed/crisp --- so it only \textbf{ties}.
  \end{block}
\end{frame}

\begin{frame}{Set B: the formula --- how fuzzy ``talks to'' the policy}
  \textbf{Key point:} fuzzy is \textbf{not inside the policy}. The policy is plain PPO --- fuzzy only
  \textbf{picks the next training soil}. One loop per training block:
  \vspace{2pt}
  \begin{enumerate}\footnotesize
    \item \textbf{Train} the policy (PPO) on the current friction rung $\mu_{\text{rung}}$.
    \item \textbf{Score skill:} \;$m=\dfrac{\overline{\text{ep\_len}}}{\text{cap}}\in[0,1]$ \quad(survives longer $\Rightarrow m\to1$).
    \item \textbf{Fuzzy $\to$ advance score} $a\in[0,1]$ (smooth Hold\,/\,Maybe\,/\,Advance):
      \[\;a=\frac{0\cdot\mu_{\text{Low}}+0.5\cdot\mu_{\text{Med}}+1\cdot\mu_{\text{High}}}
              {\mu_{\text{Low}}+\mu_{\text{Med}}+\mu_{\text{High}}}\;\]
      \vspace{-2pt}\scriptsize ($\mu_{\text{Low/Med/High}}=$ how much skill $m$ counts as low / medium / high).
    \item \footnotesize \textbf{Accumulate \& advance:} progress $\mathrel{+}{=}\,a$; \;if progress $\ge 1$
      $\Rightarrow$ move to \textbf{harder} soil, progress $\mathrel{-}{=}1$.
  \end{enumerate}
  \begin{block}{The 3 arms differ \textit{only} in steps 3--4}
    \footnotesize \textbf{fixed:} advance on a timer (ignore $m$). \;\;
    \textbf{crisp:} advance iff $m\ge0.5$ (hard yes/no). \;\;
    \textbf{fuzzy:} accumulate the graded $a$ above.
  \end{block}
\end{frame}

\begin{frame}{The catch behind the pretty chart}
  The $\rho\approx0.97$ looks great --- but it is \textbf{only the traction (friction) axis}.
  \begin{itemize}
    \item In rigid PhysX the \textbf{support/firmness axis does nothing} (feet can't sink).
    \item So $d$ is really just a \textbf{re-labelling of friction} $\mu$ --- it carries
          \bad{no information that $\mu$ doesn't already give}.
    \item A 2-input fuzzy model can't earn its keep when \textbf{one input is dead}.
  \end{itemize}
  \begin{block}{So we need\dots}
    a simulator where soil \textbf{firmness is physically real} --- next.
  \end{block}
\end{frame}

\begin{frame}{The fix: DFH deformable soil}
  \textbf{DFH} (Deformable Furrowed Heightfield) adds the missing physics:
  feet \textbf{sink} (Bekker model: compaction under the foot), generate \textbf{sideways drag},
  and the soil \textbf{deforms}.
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \centering \fig{dfh_support_axis.png}{\linewidth}
    \end{column}
    \begin{column}{0.48\textwidth}
      \good{The support axis is now real:}
      \begin{itemize}
        \item Soft soil ($K{=}1$): feet sink $\sim$\textbf{49\,mm}.
        \item Firm soil ($K{=}8$): sink $\sim$\textbf{23\,mm}; drag drops $\sim$\textbf{45$\to$20}\,N.
        \item Confirmed by a force-off control (no forces $\Rightarrow$ no effect).
      \end{itemize}
      \textit{Same robot, same gait — only the soil firmness changes. You can see it.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Set C: the key result}
  Now \textbf{both} soil dials are physical. Does the fuzzy index finally add value?
  \begin{columns}
    \begin{column}{0.52\textwidth}
      \centering \fig{dfh_fuzzy_vs_raw.png}{\linewidth}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item Raw soil numbers \good{strongly predict} difficulty (+0.33 to +0.42).
        \item The paper's \textbf{fuzzy index} adds \bad{+0.025} — essentially nothing
              (statistically detectable, $p{=}0.001$, but practically zero).
      \end{itemize}
      \begin{block}{Sharp finding}
        Even where firmness \textbf{is} physically active, the fuzzy mapping \textbf{fails to use it}.
        The problem is the \textbf{fuzzy model's structure}, \textbf{not} the simulator.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Giving fuzzy its best shot: noisy soil sensing}
  Fuzzy logic's real strength is \textbf{tolerating imprecise inputs}. A real robot only
  \textit{estimates} the soil. So we added sensing noise.
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \centering \fig{fuzzy_noise_analysis.png}{\linewidth}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{itemize}
        \item Fuzzy \textbf{is} noise-stable (flat line) and edges raw on error-\emph{size}
              \textit{only} at extreme noise.
        \item But it ``wins'' by being \bad{stable, not accurate} — even there, raw still
              \textbf{orders soils better}. Flatness $\ne$ usefulness.
        \item \textbf{Decisions} (\textcircled{2}): a fuzzy-driven curriculum \bad{camps on easy soil}
              and never advances.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \footnotesize Stability of an \textit{uninformative} signal — it doesn't swing because it barely says anything.
\end{frame}

\begin{frame}{Set D: can we \textit{fix} the broken fuzzy rules?}
  Set C blamed the \textbf{rule structure} (it ignores firmness). So we redesigned it ---
  \textbf{only} the rule table, so support now matters at \textit{every} traction level.
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \centering
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{@{}lc@{}}
        \toprule
        \textbf{Descriptor} & \textbf{Uses firmness?} \\
        \midrule
        paper fuzzy index         & \bad{+0.025} (blind) \\
        \textbf{redesigned fuzzy} & \good{+0.416} (sees it) \\
        raw soil numbers          & +0.33 to +0.42 \\
        \bottomrule
      \end{tabular}\\[4pt]
      \footnotesize Cross-validated $R^2$ gain over friction alone (higher = uses firmness more).
      The redesign lands \textbf{inside} the raw numbers' range --- it \textit{matches} them, not beats.
    \end{column}
    \begin{column}{0.48\textwidth}
      \good{The fix works — as a descriptor:}
      \begin{itemize}
        \item Firmness signal jumps \textbf{0.025 $\to$ 0.416} ($\sim$16$\times$).
        \item The original failure was \textbf{structural and fixable}, not the simulator.
      \end{itemize}
      \footnotesize \textit{Honest:} the fix is informed by the data's firm=easier trend —
      a \textbf{corrected} descriptor, not a vindication of the paper's rules. Raw soil
      numbers stay the ceiling; the fix only \textbf{matches} them, never exceeds.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Set D: \emph{exactly} what we changed in the rules}
  Same 2 inputs, same fuzzy membership shapes, same defuzzification --- we changed \textbf{only
  the rule lookup} (grip, firmness)$\,\to\,$difficulty, plus the output detail (\textbf{3$\to$5} levels).
  \footnotesize\textit{(rows $=$ grip/traction, columns $=$ firmness)}\normalsize
  \vspace{3pt}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \centering
      \bad{Paper rules} \;\scriptsize(Easy\,/\,Mod\,/\,Hard)\normalsize\\[3pt]
      \renewcommand{\arraystretch}{1.3}\small
      \begin{tabular}{@{}l|ccc@{}}
         & Soft & Med & Firm \\ \midrule
        Low  & \bad{Hard} & \bad{Hard} & \bad{Hard} \\
        Mid  & \bad{Mod}  & \bad{Mod}  & \bad{Mod}  \\
        High & Mod  & Easy & Easy \\
      \end{tabular}\\[5pt]
      \footnotesize \bad{Low \& Mid rows are flat}: firmness changes \textbf{nothing} $\Rightarrow$
      the index is \textbf{blind} to how soft the soil is.
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \good{Redesigned rules (v2)} \;\scriptsize(VE\,$<$\,E\,$<$\,M\,$<$\,H\,$<$\,VH)\normalsize\\[3pt]
      \renewcommand{\arraystretch}{1.3}\small
      \begin{tabular}{@{}l|ccc@{}}
         & Soft & Med & Firm \\ \midrule
        Low  & \bad{VH} & H & M  \\
        Mid  & H  & M & E  \\
        High & M  & E & \good{VE} \\
      \end{tabular}\\[5pt]
      \footnotesize \good{Every row falls Soft$\to$Firm}: firmer soil $=$ easier, at
      \textbf{every} grip level.
    \end{column}
  \end{columns}
  \vspace{3pt}
  \begin{block}{Payoff}
    Firmness signal jumps \bad{+0.025} $\to$ \good{+0.416} ($\sim$16$\times$) --- \textbf{purely}
    from a better rule table. The broken part was \textbf{structural}; the fix needs \textbf{no new physics}.
  \end{block}
\end{frame}

\begin{frame}{Does the redesigned fuzzy index actually work? (descriptor check)}
  \centering
  \fig{setD_fuzzy_predicts.png}{0.58\linewidth}\\[2pt]
  \footnotesize Within the measured sweep, the redesigned fuzzy index \good{orders real difficulty}

\begin{frame}{Set D: but does the fix help the robot \textit{train}?}
  \textbf{The catch:} a better \textit{description} of soil is not the same as a better
  \textit{training signal}. We pace the curriculum (when to move to harder soil) with the fixed
  index --- cheap gated proxy, vetted with an independent model.
  \begin{itemize}
    \item As a single score: \bad{fails} — it trails the raw numbers and thrashes.
    \item In its \textbf{best form} $[\mu,\text{fuzzy}]$: \good{fully fixes} the camping and
          \textbf{matches} the raw numbers' accuracy\dots
    \item \dots but only \textbf{ties} them --- it carries the \bad{same information} as raw
          $[\mu,\log_2 K]$, never beats it (a pre-registered \textbf{equivalence} result: a tie, not a win).
  \end{itemize}
  \begin{block}{Decision}
    \textbf{GPU training run not justified} ($\sim$24\,h saved). A better \textit{description}
    of the soil does \textbf{not} become a better \textit{training signal} than the raw numbers.
  \end{block}
\end{frame}

\begin{frame}{Set D in one picture: a descriptor win, not a training win}
  \centering
  \fig{setD_descriptor_vs_training.png}{0.96\linewidth}\\[3pt]
  \footnotesize Fixing the fuzzy \textbf{rule table} makes the index physically meaningful as a
  \good{descriptor} (left: $0.025\to0.416$, reaching the raw/physics ceiling) --- but as a

\end{frame}

\begin{frame}{Verdict}
  \begin{block}{Bottom line}
    Across rigid \textbf{and} deformable soil, the paper's fuzzy soil index shows
    \textbf{no demonstrable value} over the raw soil numbers — as a label, a curriculum,
    a descriptor, or under noisy sensing. And when we \textbf{fix} it, it becomes a good
    descriptor but \textbf{still only ties} raw for training.
  \end{block}
  \begin{itemize}
    \item \good{Replicates:} the friction-curriculum \textbf{survival} benefit; DFH genuinely
          restores the firmness axis.
    \item \bad{Does not add value:} the paper's mapping collapses two soil factors into a coarse
          score and \textbf{discards the useful information}.
    \item The failure is \textbf{structural} (Set C), and \textbf{fixable} (Set D) — but the fix
          buys \textbf{interpretability, not performance}: raw is the ceiling fuzzy can reach,
          not exceed.
  \end{itemize}
\end{frame}

\begin{frame}{Scope \& what would change the answer}
  \textbf{Honest scope:}
  \begin{itemize}
    \item This is about \textbf{this paper's specific fuzzy mapping} — \textit{not} all 2-D or
          learned terrain descriptors.
    \item DFH is an \textbf{uncalibrated} deformable-soil surrogate; one frozen walker.
    \item Not a disproof of the paper on \textbf{real} soil — a non-replication \textit{in simulation}.
  \end{itemize}
  \textbf{What could make ``fuzzy matter'':}
  \begin{itemize}
    \item \sout{Redesign the rules to use the firmness axis} — \good{done (Set D)}: it works as a
          \textbf{descriptor} but still only \textbf{ties} raw for training.
    \item A genuinely \textbf{new} claim with \textbf{noise-stability as the goal}, or a setting
          where the raw soil numbers \textbf{aren't available} (e.g.\ a real robot with only a
          coarse visual soil label, no precise sensors).
    \item Train a soil-robust walker (removes the single-policy caveat).
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \centering
  \Large \textbf{A fuzzy score is only useful if it beats the numbers it's made from.}\\[10pt]
  \normalsize Here it doesn't — even after we fixed it, the best it does is \textbf{tie}.\\[8pt]
  \footnotesize \textit{Fuzzy soil difficulty is a strong, interpretable \textbf{description} of
  terrain hardness, but it does not improve curriculum \textbf{performance} over the raw soil
  parameters for Hunter locomotion.}\\[14pt]
  \footnotesize
  Code \& full write-ups: \texttt{docs/experiments/fuzzy\_soil\_UNIFIED\_conclusion.md};
  models on Hugging Face \texttt{anhrisn/hunter-dfh-locomotion}.
\end{frame}
